# 課題4-1

```js
// 外部APIを模した関数（後でモックする用）
import axios from 'axios';

/**
 * ユーザー情報を外部APIから取得する関数
 */
export async function fetchUserData(userId: string): Promise<{ id: string; name: string }> {
  const response = await axios.get(`https://api.example.com/users/${userId}`);
  return response.data;
}

/**
 * メールアドレスを検証して結果を返す（バリデーションロジック）
 */
export function isValidEmail(email: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

/**
 * 商品価格と税率から税込価格を計算し、ログを出す
 */
export function calculateTax(price: number, taxRate: number, logger: (message: string) => void): number {
  const total = price * (1 + taxRate);
  logger(`Calculated tax: ${total}`);
  return total;
}
```
