order {
  bigint id PK "ID"
  varchar name "お名前"
  varchar phone_number "電話番号"
  boolean isPaid "支払い済み"
  boolean isTakeOut "持ち帰り"
  varchar requests "その他"
}

menu {
  bigint id PK "ID"
  varchar menu_name "メニュー名"
  number price "値段" 
  bigint set_menu_id FK "セットメニューID"
}

set_menu{
	bigint id PK "ID"
	bigint order "順番"
	bigint category_id FK "カテゴリーID"
}

set_menu_category {
	bigint id PK "ID"
	varchar name "カテゴリー名"
}

ingredient {
  bigint id PK "ID"
  varchar name "材料名"
}

menu_ingredient {
  bigint menu_id FK "メニューID"
  bigint ingredient_id FK "材料ID"
}

order_detail {
	bigint order_id PK "注文ID"
	bigint order_id FK "注文ID"
	bigint menu_id PK "メニューID"
	bigint menu_id FK "メニューID"
	number count "個数"
	boolean excludesWasabi "さび抜き"
	boolean isSmallRice "シャリが小さいか"
}
